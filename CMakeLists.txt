cmake_minimum_required(VERSION 3.20)
project(auto-apo-switch LANGUAGES CXX RC)
add_executable(AutoAPO WIN32 main.cpp AutoAPO.rc)
target_include_directories(AutoAPO PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(AutoAPO PRIVATE cxx_std_17)
target_compile_definitions(AutoAPO PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(AutoAPO PRIVATE comctl32 comdlg32 shell32 advapi32 ole32 oleaut32)
if(MSVC)
  target_compile_options(AutoAPO PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/utf-8>")
  set_property(TARGET AutoAPO PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
